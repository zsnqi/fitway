			/* P11_NATIVE_CAUSAL_BEGIN: scratch-only wrapper of the original await. */
			try {

				"changed-count",

				const causalEvidence: Record<string, any> = {

					dbReads: 0,

					if (!condition) throw new Error(code);

								exits: w1Count(minute.exits),
							}))

					sequence: w1Count(state.sequence),

						"SELECT cs.current_count AS \"currentCount\", " +

						"ed.last_sequence::int AS \"lastSequence\", " +
						"(SELECT COALESCE(json_agg(h ORDER BY h.\"minuteStart\"), '[]'::json) FROM " +
						"(SELECT m.minute_start_utc AS \"minuteStart\", m.count, m.entries, m.exits, m.source, " +
						"m.updated_at AS \"updatedAt\" FROM occupancy_minutes m WHERE m.device_id=ed.id " +
						"ORDER BY m.minute_start_utc LIMIT 4) h) AS history " +
						"FROM current_state cs JOIN edge_devices ed ON ed.id=cs.active_device_id WHERE cs.id=1",
					);
					requireCausal(result.rows.length === 1, "P11_DATABASE_SINGLETON_MISSING");
					const row = result.rows[0];
					return {
						observedElapsedMs: Date.now() - diagnosticStartedAt,
						currentCount: w1Count(row.currentCount),
						lastSequence: w1Count(row.lastSequence),

						lastEdgeReportedAt: isoCausal(row.lastEdgeReportedAt),
						history: row.history.map((minute: any) => ({
							minuteStart: isoCausal(minute.minuteStart),
							count: w1Count(minute.count),
							entries: w1Count(minute.entries),
							exits: w1Count(minute.exits),
							source: w1Token(minute.source, ["live", "backfill", "manual"]),

						countText: document.querySelector(".public-live__count-value")?.textContent?.trim().slice(0, 24) ?? null,

				let restoreCausalRecord: (() => void) | undefined;
				let causalDeadlineTimer: ReturnType<typeof setTimeout> | undefined;
				try {

							originalChangedFailure.name === "TimeoutError" &&

					requireCausal(changedCount !== firstCount && beforeState.count === changedCount && beforeState.sequence === 2 && beforeState.outboxLength === 0 && beforeState.inFlightEmpty, "P11_SECOND_PERSISTED_STATE_MISMATCH");
